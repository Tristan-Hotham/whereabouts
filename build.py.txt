def lad(lon,lat):
    pt=Point(lon,lat)
    for i in tree.query(pt):
        if polys[i].contains(pt): return codes[i]
    # nearest (coastal points)
    j=min(range(len(polys)),key=lambda i:polys[i].distance(pt)); return codes[j] if polys[j].distance(pt)<0.05 else None
def km(a,b,c,d):
    R=6371;p1,p2=math.radians(a),math.radians(c);dp=p2-p1;dl=math.radians(d-b)
    return 2*R*math.asin(math.sqrt(math.sin(dp/2)**2+math.cos(p1)*math.cos(p2)*math.sin(dl/2)**2))

def pct(arr,v): return round(100*sum(1 for x in arr if x<v)/len(arr))
FUNIDS=[k for k,c in cats.items() if c['group']=='Just for fun' and k not in ('english','irishpass','walk','threecars')]

def rank_of(k,code):
    v=cats[k]['v']; ks=sorted(v,key=lambda x:-v[x]); return ks.index(code)+1,len(ks),v[code]
def quirk(code):
    def fun(lim):
        best=None
        for k in FUNIDS:
            v=cats[k]['v']
            if code not in v or v[code]==0: continue
            r,n,val=rank_of(k,code)
            if r<=n*lim and (best is None or r/n<best[0]): best=(r/n,k,r,n,val,'hi')
        return best
    def ser(lim):
        best=None
        for k in SERIOUS:
            v=cats[k]['v']
            if code not in v: continue
            r,n,val=rank_of(k,code)
            for frac,side,rr in ((r/n,'hi',r),((n-r+1)/n,'lo',n-r+1)):
                if frac<=lim and (best is None or frac<best[0]): best=(frac,k,rr,n,val,side)
        return best
    return fun(0.05) or ser(0.05) or fun(0.10) or ser(0.10) or fun(0.15)
def fmt(c,x):
    f=c['fmt']
    if f=='gbp': return '£{:,.0f}'.format(x)
    if f=='pct': return ('%.1f%%'%x) if c['dp']==1 else ('%.2f%%'%x)
    if f=='int': return '{:,.0f}'.format(x)
    return ('%.1f'%x)
def ordn(n): return str(n)+('th' if 10<=n%100<=20 else {1:'st',2:'nd',3:'rd'}.get(n%10,'th'))

def meaning(nm):
    if nm in SPECIAL: return SPECIAL[nm]
    if nm in SKIP: return None
    w=re.sub(r'[^a-z ]','',nm.lower()).split()
    first,last=(w[0],w[-1]) if w else ('','')
    for p,m in PRE:
        if first.startswith(p) and len(first)>len(p): return m
    for sfx,m in SUF:
        if last.endswith(sfx) and len(last)>len(sfx)+1: return m
    return None

def bearing(a,b,c,d):
    y=math.sin(math.radians(d-b))*math.cos(math.radians(c));x=math.cos(math.radians(a))*math.sin(math.radians(c))-math.sin(math.radians(a))*math.cos(math.radians(c))*math.cos(math.radians(d-b))
    return (math.degrees(math.atan2(y,x))+360)%360
def council_clues(c,cl):
    name,nat,label=places[c]
    bits=[]
    hp=cats['house']['v'].get(c); lv=cats['leave']['v'].get(c); pay=cats['pay']['v'].get(c)
    if hp: bits.append(f"the median house costs £{hp:,.0f}")
    elif pay: bits.append(f"median full-time pay is £{pay:,.0f}")
    if lv: bits.append(f"{lv:.0f}% voted Leave in 2016")
    cl.append("In its council area "+" and ".join(bits)+"." if bits else "Its council area is one of the UK's 359.")
    q=quirk(c)
    if q:
        _,k,r,n,v,side=q; cc=cats[k]
        lab=cc['label'].replace('Life expectancy (women)','life expectancy for women').replace('Life expectancy (men)','life expectancy for men') if cc['label'].startswith('Life') else cc['label'] if cc['label'].split()[0] in ('Queen',"Queen's",'Pagans,','English,','Irish') else cc['label'][0].lower()+cc['label'][1:]
        val=fmt(cc,v)+('' if cc['fmt'] in ('pct','gbp') else ' '+cc['unit'])
        if side=='hi' and cc['group']=='Just for fun': cl.append(f"Fun fact: its council area ranks {ordn(r)} of {n} for {lab} ({val}).")
        elif side=='hi': cl.append(f"Fun fact: its council area has the {'' if r==1 else ordn(r)+' '}highest {lab} of {n} councils ({val}).".replace('the  ','the '))
        else: cl.append(f"Fun fact: its council area has the {'' if r==1 else ordn(r)+' '}lowest {lab} of {n} councils ({val}).".replace('the  ','the '))
        return True
    return False
def where_clue(c):
    name,nat,label=places[c]
    if ' · ' in label: return f"It's in {label.split(' · ')[0]}, {label.split(' · ')[1]}."
    if label=='London': return "It's in London."
    return f"Its council is {name}, in {label}."
def letters(nm):
    L=len(re.sub(r'[^A-Za-z]','',nm))
    return f"starts with {nm[0]} and has {L} letters"+(" (not counting spaces and hyphens)" if re.search(r'[^A-Za-z]',nm) else "")
def make(t,c,kind,sun,rain,jul,sunpool,daily,extra6=None):
    name,nat,label=places[c]
    nm=t['n']; p=t['pop']; pr=int(float('%.2g'%p)) if p else None
    cl=[f"A {kind} of about {pr:,} people in {NAT[nat]}." if pr else f"A {kind} in {NAT[nat]}."]
    cl.append(f"It gets about {sun:,.0f} hours of sunshine and {rain:,.0f} mm of rain a year: sunnier than {pct(sunpool,sun)}% of UK {kind}s.")
    if not council_clues(c,cl): cl.append(f"Fun fact: July afternoons there average {jul:.1f} °C, warmer than {pct(julr,jul)}% of UK towns.")
    cl.append(where_clue(c))
    cl.append(extra6 or ("It "+letters(nm)+"."))
    return dict(d=daily,m=meaning(nm),n=nm,lat=round(t['lat'],4),lon=round(t['lon'],4),c=c,area=label,council=name,clues=cl,pop=pr)
