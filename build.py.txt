import csv,json,math,re
from collections import defaultdict
from shapely.geometry import shape,Point
from shapely import STRtree
P='repos/'
H=json.load(open('higher-or-lower/data.json'))
places=H['places']; cats={c['id']:c for c in H['cats']}
geo=json.load(open(P+'footpaths/data/lad.geojson'))
polys=[shape(f['geometry']) for f in geo['features']]; codes=[f['properties']['code'] for f in geo['features']]
tree=STRtree(polys)
def lad(lon,lat):
    pt=Point(lon,lat)
    for i in tree.query(pt):
        if polys[i].contains(pt): return codes[i]
    # nearest (coastal points)
    j=min(range(len(polys)),key=lambda i:polys[i].distance(pt)); return codes[j] if polys[j].distance(pt)<0.05 else None
def km(a,b,c,d):
    R=6371;p1,p2=math.radians(a),math.radians(c);dp=p2-p1;dl=math.radians(d-b)
    return 2*R*math.asin(math.sqrt(math.sin(dp/2)**2+math.cos(p1)*math.cos(p2)*math.sin(dl/2)**2))
# towns (sun-and-rain): dedupe, prefer GeoNames rows with population
rows=list(csv.DictReader(open(P+'sun-and-rain-uk/data/towns.csv')))
from pyproj import Transformer
_tr=Transformer.from_crs(27700,4326,always_xy=True)
for r in rows:
    lo,la=_tr.transform(float(r['e']),float(r['n'])); r['lat'],r['lon']=str(la),str(lo)
rows.sort(key=lambda r:(r['source']!='GeoNames'))
CITIES={r['name'] for r in rows if r['kind']=='City'}
towns=[]
for r in rows:
    lat,lon=float(r['lat']),float(r['lon'])
    if any(t['n']==r['name'] and km(lat,lon,t['lat'],t['lon'])<15 for t in towns): continue
    towns.append(dict(n=r['name'],lat=lat,lon=lon,kind=r['kind'],pop=float(r['population']) if r['population'] else None,
        sun=float(r['sun']),rain=float(r['rain']),jul=float(r['tmax_Jul']),nation=r['nation']))
# BUA populations (E&W + GB) for better numbers and extra guessable places
bua=list(csv.DictReader(open(P+'postbox-kings/data/towns.csv')))
for t in towns:
    best=None
    for b in bua:
        if b['BUA22NM']==t['n'] or b['BUA22NM'].split(' (')[0]==t['n']:
            d=km(t['lat'],t['lon'],float(b['LAT']),float(b['LONG']))
            if d<10 and (best is None or d<best[0]): best=(d,float(b['pop2021'] or 0))
    if best and best[1]>0: t['pop']=best[1]; t['bua']=True
print('towns',len(towns))
# guess list: towns + BUAs over 2,000 people not already listed
guess=[(t['n'],t['lat'],t['lon']) for t in towns]
have=defaultdict(list)
for n,la,lo in guess: have[n].append((la,lo))
for b in bua:
    n=b['BUA22NM'].split(' (')[0]; la,lo=float(b['LAT']),float(b['LONG'])
    if float(b['pop2021'] or 0)<2000: continue
    if any(km(la,lo,x,y)<10 for x,y in have[n]): continue
    guess.append((n,la,lo)); have[n].append((la,lo))
print('guessable',len(guess))
# label each guess with its council's area label for disambiguation
G=[]
for n,la,lo in guess:
    c=lad(lo,la)
    if not c or c not in places: continue
    G.append([n,round(la,4),round(lo,4),c])
# names that repeat get a hint
cnt=defaultdict(int)
for g in G: cnt[g[0]]+=1
# answer pool
sunr=sorted(t['sun'] for t in towns); julr=sorted(t['jul'] for t in towns)
def pct(arr,v): return round(100*sum(1 for x in arr if x<v)/len(arr))
FUNIDS=[k for k,c in cats.items() if c['group']=='Just for fun' and k not in ('english','irishpass','walk','threecars')]
SERIOUS=[k for k,c in cats.items() if c['group']!='Just for fun' and k not in ('pop','area','density')]
def rank_of(k,code):
    v=cats[k]['v']; ks=sorted(v,key=lambda x:-v[x]); return ks.index(code)+1,len(ks),v[code]
def quirk(code):
    def fun(lim):
        best=None
        for k in FUNIDS:
            v=cats[k]['v']
            if code not in v or v[code]==0: continue
            r,n,val=rank_of(k,code)
            if r<=n*lim and (best is None or r/n<best[0]): best=(r/n,k,r,n,val,'hi')
        return best
    def ser(lim):
        best=None
        for k in SERIOUS:
            v=cats[k]['v']
            if code not in v: continue
            r,n,val=rank_of(k,code)
            for frac,side,rr in ((r/n,'hi',r),((n-r+1)/n,'lo',n-r+1)):
                if frac<=lim and (best is None or frac<best[0]): best=(frac,k,rr,n,val,side)
        return best
    return fun(0.05) or ser(0.05) or fun(0.10) or ser(0.10) or fun(0.15)
def fmt(c,x):
    f=c['fmt']
    if f=='gbp': return '£{:,.0f}'.format(x)
    if f=='pct': return ('%.1f%%'%x) if c['dp']==1 else ('%.2f%%'%x)
    if f=='int': return '{:,.0f}'.format(x)
    return ('%.1f'%x)
def ordn(n): return str(n)+('th' if 10<=n%100<=20 else {1:'st',2:'nd',3:'rd'}.get(n%10,'th'))
def nat_of(t): return {'England':'E','Wales':'W','Scotland':'S','Northern Ireland':'N'}[t['nation']]

def meaning(nm):
    if nm in SPECIAL: return SPECIAL[nm]
    if nm in SKIP: return None
    w=re.sub(r'[^a-z ]','',nm.lower()).split()
    first,last=(w[0],w[-1]) if w else ('','')
    for p,m in PRE:
        if first.startswith(p) and len(first)>len(p): return m
    for sfx,m in SUF:
        if last.endswith(sfx) and len(last)>len(sfx)+1: return m
    return None
