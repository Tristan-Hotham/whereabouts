for c,(n,nat,l) in places.items():
    if c not in G: continue

        if ' · ' in l: u,r=l.split(' · ')

agg={}
for k,cc in cats.items():
    v=cc['v'];out={}
    for u,cs in mem.items():
        if not all(c in v for c in cs): continue
        if k in COUNT: out[u]=sum(v[c] for c in cs)
        else:
            w=[(area if k in AREAW else pop).get(c,0) for c in cs]
            if sum(w)==0: continue
            out[u]=sum(v[c]*x for c,x in zip(cs,w))/sum(w)
    agg[k]=out
# order so pop/area exist before density
agg['density']={u:agg['pop'][u]/agg['area'][u] for u in agg['pop'] if u in agg['area']}
FUN=[k for k,c in cats.items() if c['group']=='Just for fun' and k not in ('english','irishpass','walk','threecars')]
SER=[k for k,c in cats.items() if c['group']!='Just for fun' and k not in ('pop','area','density','sun','rain')]
def rk(k,u):
    v=agg[k]; ks=sorted(v,key=lambda x:-v[x]); return ks.index(u)+1,len(ks),v[u]
def quirk(u):
    def fun(lim):
        b=None
        for k in FUN:
            if u not in agg[k] or agg[k][u]==0 or len(agg[k])<40: continue
            r,n,val=rk(k,u)
            if r<=max(1,n*lim) and (b is None or r/n<b[0]): b=(r/n,k,r,n,val,'hi')
        return b
    def ser(lim):
        b=None
        for k in SER:
            if u not in agg[k] or len(agg[k])<40: continue
            r,n,val=rk(k,u)
            for f,s,rr in ((r/n,'hi',r),((n-r+1)/n,'lo',n-r+1)):
                if rr<=max(1,n*lim) and (b is None or f<b[0]): b=(f,k,rr,n,val,s)
        return b
    return fun(.03) or ser(.03) or fun(.06) or ser(.06) or fun(.1) or ser(.1)
def fmt(c,x):
    f=c['fmt']
    if f=='gbp': return '£{:,.0f}'.format(x)
    if f=='pct': return '%.1f%%'%x
    if f=='int': return '{:,.0f}'.format(x)
    return '%.1f'%x if c['dp']<=1 else '%.2f'%x
def ordn(n): return str(n)+('th' if 10<=n%100<=20 else {1:'st',2:'nd',3:'rd'}.get(n%10,'th'))
def lab(cc):
    l=cc['label']
    l={'No passport':'people with no passport','No religion':'people with no religion','Very good health':'people in very good health','No central heating':'homes with no central heating','Retired':'retired people','Self-employed':'self-employed people','Area':'area','Happiness':'happiness ratings','Life satisfaction':'life satisfaction ratings'}.get(l,l)
    if l.startswith('Life'): return l.replace('Life expectancy (women)','life expectancy for women').replace('Life expectancy (men)','life expectancy for men')
    return l if l.split()[0] in ('Queen',"Queen's",'Pagans,','English,','Irish') else l[0].lower()+l[1:]

closed=land.buffer(0.02).buffer(-0.02)

def THE(r): return 'the '+r if r in ('East Midlands','West Midlands','East of England') else r
def letters(nm):
    L=len(re.sub(r'[^A-Za-z]','',nm)); return f"starts with {nm[0]} and has {L} letters"+(" (not counting spaces and punctuation)" if re.search(r'[^A-Za-z]',nm) else "")
sunv=sorted(agg['sun'].values())
def pct(a,v): return round(100*sum(1 for x in a if x<v)/len(a))
def roundp(p): return int(float('%.2g'%p))
